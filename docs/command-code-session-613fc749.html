- `language`：level 0 → '看不见'；level 1、2 → '选一句'；level 3 → '自由说'。
- `slot`：wait < 0.4 → '现在'；否则 polish > 0.6 → '稍后'，否则 '现在'。
- `why`：一句话，说出依据，必须提到 brief 里的一个字段（例：“你说想要能真的改变它，所以这里放开”）。
`director.design` 写一条 `director.design`，`payload = design`，`cause` 包含传入的 cause。

`predict`：对每个 level ≥ 1 的 beat，用导演心智的 `imagine` 设想玩家（玩家性情由 notes 推：depth→好奇，surprise→冒险，polish→雕琢），预测 expect ∈ `'停下来看心思'|'追问为什么'|'很快通过'|'重来一次'`，confidence 0..1。映射规则：设想结果 stoppedAt==='mind' → '停下来看心思'；'module' → '追问为什么'；'unit' → '很快通过'。

`learn`：reactions = `[{ beat, signal }]`。每条与预测对照：相同 error 0，不同 error 1。整体命中率低于 0.5 时：若玩家多停下来看心思而导演预测很快通过 → `paceBias -0.1`、`tragedyBias +0.05`；反之 `paceBias +0.1`。每次 learn 写一条 `director.learn`（含 craft.before/after）。手艺参数会影响下一次 design 的 level 选择（paceBias > 0.15 时 level 往低调 1）。

export function createDirector({ ledger, def, mentor }) {
  const mind = createMind({ units: [], modules: [], temperament: {}, ...def, id: def.id });
  const role = 'director:' + def.id;
  const t = def.temperament || {};
  // 手艺参数：从导演性情起步，learn 会改它
  const craft = {
    lockBias: round(num(t.雕琢, 0.5)),
    tragedyBias: round(num(t.悲剧, 0.5)),
    paceBias: 0,
    freedomBias: round(num(t.放开, 0.5)),
  };

  const append = (kind, payload, cause, said, saidSource) => {
    const e = ledger.append({
      kind, from: role, to: 'world', payload,
      cause: causes(cause), ...(said !== undefined ? { said, saidSource } : {}),
    });
    return e.seq;

  // 模板：emotions 里与 brief.emotion 相同的优先，否则 emotions[0]，再查 TEMPLATES
  function templateOf(beat, emotion) {
    const emos = beat.emotions || [];
    const emo = emos.includes(emotion) ? emotion : emos[0];
    if (TEMPLATES[emo]) return TEMPLATES[emo][0];
    const hit = Object.keys(TEMPLATES).find((k) => emos.includes(k));
    return hit ? TEMPLATES[hit][0] : '分配';
  }

  function levelOf(beat, polish) {
    const levels = [...(beat.levels || [0])].sort((a, b) => a - b);
    const lo = levels[0];
    const hi = levels[levels.length - 1];
    let level = polish > 0.6 ? lo : polish < 0.4 ? hi : levels[Math.floor((levels.length - 1) / 2)];
    if (craft.freedomBias > 0.6) level = Math.min(level + 1, hi);
    if (craft.paceBias > 0.15) level = Math.max(level - 1, 0);
    return clamp(level, 0, 3);
  }

  function grantsOf(template, level) {
    let g;
    if (level === 0) g = [];
    else if (template === '即时动作') g = ['go', 'give'];
    else if (template === '观看') g = [];
    else if (level >= 2) g = ['go', 'give', 'say'];
    else g = ['say'];
    if (level >= 1 && !g.includes('watch')) g.push('watch');
    return g;

    const prefs = brief?.prefs || {};
    const pol = num(prefs.polish);

    const emotion = brief?.emotion;
    const slot = wait < 0.4 ? '现在' : pol > 0.6 ? '稍后' : '现在';

      const prefab = locked ? '高' : level >= 2 ? '低' : '中';
      const grants = grantsOf(template, level);

      : '这一场我没有可猜的地方。';
    const seq = append('director.predict', { presentation: design?.presentation, predictions }, cause, said, 'rule');

    const verbs = (beat.grants || []).map((v) => VERB_TEXT[v] || v);
    const said = verbs.length
      ? `这一幕我给你的是：${verbs.join('、')}。`
      : '这一幕你只能看着，不能动。';

      scene: beat.scene, verbs: beat.grants || [], level: beat.level, language: beat.language,
    }, cause, said, 'rule');

    const said = chosen ? `这一次我先排「${chosen.id ?? chosen.label ?? '这一段'}」，${slot}做。` : '这次没有可排的。';
    const seq = append('director.schedule', {
      input: list.map((o) => o.id ?? o.label ?? null), options: list, chosen, slot,
    }, cause, said, 'rule');
    return { chosen, seq };
  }

  function learn(predictions, reactions, cause) {
    const before = { ...craft };
    const errors = [];
    let hits = 0;
    let pauseMiss = 0;
    let otherMiss = 0;
    for (const rx of reactions || []) {
      const p = (predictions || []).find((x) => x.beat === rx.beat);
      const miss = !p || p.expect !== rx.signal;
      if (!miss) hits += 1;
      else if (rx.signal === '停下来看心思' && p?.expect === '很快通过') pauseMiss += 1;
      else otherMiss += 1;
      errors.push({ beat: rx.beat, expect: p?.expect ?? null, signal: rx.signal, error: miss ? 1 : 0 });
    }
    const total = (reactions || []).length;
    const hitRate = total ? round(hits / total) : 1;
    if (total && hitRate < 0.5) {
      // 玩家多停下来看心思、而导演以为很快通过 → 放慢、多写悲剧余味；否则加快
      if (pauseMiss > otherMiss) {
        craft.paceBias = round(craft.paceBias - 0.1);
        craft.tragedyBias = round(craft.tragedyBias + 0.05);
      } else {
        craft.paceBias = round(craft.paceBias + 0.1);
      }
    }
    const after = { ...craft };
    const said = total
      ? `这次我猜中了${hits}/${total}，${hitRate < 0.5 ? '偏差大，我把节奏调了' : '基本准，手艺不动'}。`
      : '这次没有反应可对照。';
    // 契约形状：beat / expected / actual / error 为并列数组，另附 hitRate 与 errors 明细
    const seq = append('director.learn', {
      beat: errors.map((e) => e.beat),
      expected: errors.map((e) => e.expect),
      actual: errors.map((e) => e.signal),
      error: errors.map((e) => e.error),
      errors, hitRate, craft: { before, after },
    }, cause, said, 'rule');
    return { errors, seq };
  }

  // 演示权限边界：导演不能改导师的规则或笔记，只能请导师的 guard 拒绝并记下
  function tryEditMentor(field) {
    return mentor ? mentor.guard(field, role) : null;
  }

  return {
    id: def.id,
    role,
    design,
    predict,
    grant,
    schedule,
    learn,
    tryEditMentor,
    craft: () => ({ ...craft }),
    state: () => ({ id: def.id, role, craft: { ...craft }, mind: mind.save() }),
  };
}

    return JSON.parse(JSON.stringify(notes));
  }

  /** 用户亲手改笔记：source 记为“你改的”，之后观察不能覆盖它 */
  function editNote(field, value, userSeq) {
    return setNote(field, value, '你改的', [userSeq], `你改成了${value}，这一条我听你的。`);
  }

  /**
   * 预算分配：depth 与 centrality 加权定档，超预算时从 centrality 最低的往下降。
   * scenes = [{ id, title, centrality }]
   */
  function allot(scenes = [], cause = []) {
    const depth = notes.depth?.value ?? DEFAULT_VALUE;
    const items = scenes.map((sc, i) => {
      const c = clamp(sc.centrality ?? DEFAULT_VALUE);
      const score = depth * 0.6 + c * 0.4;
      const tier = score >= 0.6 ? '细演' : score >= 0.35 ? '普通' : '一笔带过';
      return { id: sc.id, c, i, tier, cut: false };
    });
    const total = () => items.reduce((a, x) => a + TIER_NUM[x.tier], 0);
    while (total() > budgetNow) {
      const cands = items.filter((x) => TIER_NUM[x.tier] > 1).sort((a, b) => a.c - b.c || a.i - b.i);
      if (!cands.length) break;
      const x = cands[0];
      x.tier = x.tier === '细演' ? '普通' : '一笔带过';
      x.cut = true;
    }
    const out = {};
    for (const x of items) out[x.id] = x.tier;
    const sum = total();
    chosen = out;
    spent = sum;
    const anyCut = items.some((x) => x.cut);
    let line = depth > 0.6 ? '你说人心要看细，这几场我给细演。' : '人心看个大概就行，这几场我省着演。';
    if (anyCut) line += '预算就这么多，其余的我先一笔带过。';
    const e = write(
      'mentor.allot',
      {
        input: scenes.map((sc) => ({ id: sc.id, centrality: sc.centrality })),
        options: TIERS,
        chosen: out,
        spent: sum,
        budget: budgetNow,
      },
      [...cause, notes.depth?.seq],
      line,
    );
    allotSeq = e.seq;
    return { chosen: out, seq: e.seq };
  }

  /** 按口味配导演：polish 高偏爱雕琢，低偏爱放开；同分取先到的 */
  function chooseDirector(directors = [], cause = []) {
    const polish = notes.polish?.value ?? DEFAULT_VALUE;
    const key = polish > 0.5 ? '雕琢' : '放开';
    let best = null;
    for (const d of directors) {
      const sc = d.temperament?.[key] ?? 0;
      if (!best || sc > best.sc) best = { d, sc };
    }
    if (!best) return { director: null, seq: null };
    const id = best.d.id;
    const name = best.d.name || id;
    const e = write(
      'mentor.note',
      { field: 'director', value: id, source: '观察' },
      [...cause, notes.polish?.seq],
      `这一部我找${name}掌舵：你要的${polish > 0.5 ? '是雕琢' : '是放开'}。`,
    );
    director = { id, seq: e.seq };
    return { director: id, seq: e.seq };
  }

  /** 给导演的任务单：需要什么、什么情绪、口味、预算、档位 */
  function brief(directorId, cause = []) {
    const prefs = {};
    for (const f of FIELDS) prefs[f] = notes[f]?.value ?? DEFAULT_VALUE;
    const b = {
      director: directorId,
      need: notes.need?.value ?? '',
      emotion: notes.emotion?.value ?? '悲悯',
      prefs,
      budget: budgetNow,
      tiers: TIERS,
    };
    const seqs = FIELDS.map((f) => notes[f]?.seq);
    const e = write(
      'mentor.brief',
      b,
      [...cause, ...seqs, allotSeq],
      `我把“${b.emotion}”这个口味交给${directorId}，预算 ${budgetNow}，你说过的地方我都记着。`,
      directorId ? `director:${directorId}` : 'world',
    );
    return { brief: b, seq: e.seq };
  }

  /**
   * 从玩家的反应里学口味。只改没被用户亲手改过的字段。
   * signal = { kind, beat }。没有规则的信号（如“停顿”）返回 null。
   */
  function learn(signal = {}, userSeq) {
    const rules = LEARN_RULES[signal.kind];
    if (!rules) return null;
    let last = null;
    for (const [field, delta] of rules) {
      const cur = notes[field] ?? { value: DEFAULT_VALUE, source: '观察', seq: null };
      if (cur.source === '你改的') continue;
      const before = cur.value;
      const after = round2(clamp(before + delta));
      const e = write(
        'mentor.learn',
        { field, before, after, signal: signal.kind, beat: signal.beat },
        [userSeq],
        LEARN_SAID[signal.kind],
      );
      notes[field] = { value: after, source: '观察', seq: e.seq };
      last = e.seq;
    }
    return last;
  }

  /** 导演想改导师的规则或笔记：拒绝，并写下拒绝记录（to 指向提出者） */
  function guard(action, by) {
    // 没有可追溯的锚点时，退回账本里最近一条用户记录，保证 cause 能走到用户
    const userEntry = ledger.entries.filter((x) => x.from === 'user').pop();
    const link = anchor ?? userEntry?.seq;
    const e = write(
      'mentor.refuse',
      { attempted: action, by, reason: '导师的分配规则和笔记只听用户，导演不能改。' },
      link !== undefined ? [link] : [],
      '这是我的规则，不归导演改。',
      by,
    );
    return e.seq;
  }

  function state() {
    return JSON.parse(JSON.stringify({
      notes, chosen, spent, director, allotSeq, anchor, budget: budgetNow,
    }));
  }

  return {
    def,
    budget: budgetNow,
    meet,
    notes: snapshotNotes,
    editNote,
    allot,
    chooseDirector,
    brief,
    learn,
    guard,
    state,
  };
}

import { createLedger, loadLedger, auditService, traceToUser } from '../core/ledger.js';
import { createClaimGraph } from '../core/claims.js';

import { createMentor } from '../agents/mentor.js';

/** 史料层是设定，不由任何人的决定产生，所以记成 'world' 发出的设定主张 */
function seedHistory(claims) {
  for (const h of HISTORY) {
    claims.add({
      subject: h.subject, predicate: h.predicate, object: h.object, holder: h.holder || '史书',
      channel: '设定', evidence: h.evidence || [], weight: 3, note: h.note,
    });
  }
}

export function createSession({ saved, speaker } = {}) {
  const ledger = saved?.ledger ? loadLedger(saved.ledger) : createLedger();
  const claims = createClaimGraph(ledger);
  if (!saved) seedHistory(claims);
  const mentor = createMentor({ ledger, def: STAFF.mentor, saved: saved?.mentor });
  const directors = Object.fromEntries(STAFF.directors.map((d) => [d.id, createDirector({ ledger, def: d, mentor, saved: saved?.directors?.[d.id] })]));
  const minds = Object.fromEntries(PEOPLE.map((p) => [p.id, createMind(p, saved?.minds?.[p.id])]));
  const state = {
    met: !!saved?.met,
    directorId: saved?.directorId || null,
    designs: saved?.designs || {},
    predictions: saved?.predictions || {},
  };

  /** 第一次见面：用户回答三个粗问题，写一句自己的需求 */
  function meet(answers, need) {
    const u = ledger.append({ kind: 'user.meet', from: 'user', to: 'mentor', payload: { answers, need } });
    const r = mentor.meet({ answers, need }, u.seq);
    const pick = mentor.chooseDirector(Object.values(directors).map((d) => d.def), [u.seq, ...r.seqs]);
    state.directorId = pick.director;
    state.met = true;
    return { userSeq: u.seq, director: pick.director };
  }

  function director() { return directors[state.directorId]; }

  /**
   * 排一部作品：导师分配算力档位 → 给导演交需求 → 导演设计、授予交互、设想玩家。
   * 每一步的 cause 都接到上一步，最上面接到用户。
   */
  function commission(presentationId, cause = []) {
    const cat = CATALOG[presentationId];
    if (!cat) throw new Error(`没有这部作品：${presentationId}`);
    const root = cause.length ? cause : [ledger.last((e) => e.from === 'user')?.seq].filter(Boolean);
    const allot = mentor.allot(cat.beats.map((b) => ({ id: b.id, title: b.title, centrality: b.centrality })), root);
    const brief = mentor.brief(state.directorId, [allot.seq, ...root]);
    const d = director();
    const design = d.design({ ...brief.brief, tiers: allot.chosen }, cat, [brief.seq]);
    const grants = design.design.beats.map((b) => d.grant(b, [design.seq]));
    const notes = mentor.notes();
    const pred = d.predict(design.design, notes, [design.seq]);
    state.designs[presentationId] = { ...design.design, seq: design.seq };
    state.predictions[presentationId] = { list: pred.predictions, seq: pred.seq };
    ledger.append({ kind: 'present.open', from: `presentation:${presentationId}`, to: 'user', cause: [design.seq, ...grants], payload: { presentation: presentationId } });
    return { allot, brief, design: design.design, designSeq: design.seq, predictions: pred.predictions };
  }

  /** 用户在呈现里的反应：同时交给导师（了解你）和导演（对照预测） */
  function react(presentationId, beat, signal) {
    const u = ledger.append({ kind: 'user.react', from: 'user', to: 'mentor', payload: { scene: presentationId, beat, signal } });
    const learned = mentor.learn({ kind: signal, beat }, u.seq);
    return { userSeq: u.seq, mentorSeq: learned };
  }

  /** 一部作品看完：导演把所有反应和自己的预测对照，调整手艺 */
  function wrap(presentationId) {
    const pred = state.predictions[presentationId];
    if (!pred) return null;
    const reactions = ledger.entries
      .filter((e) => e.kind === 'user.react' && e.payload.scene === presentationId)
      .map((e) => ({ beat: e.payload.beat, signal: e.payload.signal, seq: e.seq }));
    const r = director().learn(pred.list, reactions, [pred.seq, ...reactions.map((x) => x.seq)]);
    ledger.append({ kind: 'present.deliver', from: `presentation:${presentationId}`, to: 'user', cause: [r.seq], payload: { presentation: presentationId } });
    return r;
  }

  function editNote(field, value) {
    const before = mentor.notes()[field]?.value;
    const u = ledger.append({ kind: 'user.edit-note', from: 'user', to: 'mentor', payload: { field, from: before, to: value } });
    return mentor.editNote(field, value, u.seq);
  }

  function save() {
    return {
      ledger: ledger.toJSON(),
      mentor: mentor.state(),
      directors: Object.fromEntries(Object.entries(directors).map(([k, d]) => [k, d.state()])),
      minds: Object.fromEntries(Object.entries(minds).map(([k, m]) => [k, m.save()])),
      ...state,
    };
  }

  return {
    ledger, claims, mentor, directors, minds, state, speaker,
    meet, commission, react, wrap, editNote, director, save,
    audit: () => auditService(ledger),
    trace: (seq) => traceToUser(ledger, seq),
  };
}
